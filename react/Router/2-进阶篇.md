# React Router - 进阶篇

## 嵌套路由

### 什么是嵌套路由？

嵌套路由是指在一个路由组件内部再定义子路由，常用于实现多级导航结构（如管理后台的侧边栏菜单）。

### 基本用法

```typescript
import { Routes, Route, Outlet, Link } from 'react-router-dom'

// 1. 配置嵌套路由
function App() {
  return (
    <Routes>
      <Route path="/" element={<Home />} />

      {/* 父路由 */}
      <Route path="/dashboard" element={<Dashboard />}>
        {/* 子路由 */}
        <Route path="profile" element={<Profile />} />
        <Route path="settings" element={<Settings />} />
        <Route path="orders" element={<Orders />} />
      </Route>
    </Routes>
  )
}

// 2. 父组件中使用 Outlet 渲染子路由
function Dashboard() {
  return (
    <div className="dashboard">
      <aside>
        <h2>Dashboard</h2>
        <nav>
          <Link to="/dashboard/profile">个人资料</Link>
          <Link to="/dashboard/settings">设置</Link>
          <Link to="/dashboard/orders">订单</Link>
        </nav>
      </aside>

      <main>
        <Outlet />  {/* 子路由渲染位置 */}
      </main>
    </div>
  )
}

// 3. 子组件
function Profile() {
  return <div>个人资料页面</div>
}

function Settings() {
  return <div>设置页面</div>
}

function Orders() {
  return <div>订单页面</div>
}
```

### URL 结构

```
/dashboard              → 显示 Dashboard 布局，Outlet 位置为空
/dashboard/profile      → 显示 Dashboard 布局 + Profile 组件
/dashboard/settings     → 显示 Dashboard 布局 + Settings 组件
/dashboard/orders       → 显示 Dashboard 布局 + Orders 组件
```

### 默认子路由（Index Route）

**问题**：访问 `/dashboard` 时，Outlet 位置为空

**解决方案**：使用 `index` 属性定义默认子路由

```typescript
<Route path="/dashboard" element={<Dashboard />}>
  <Route index element={<DashboardHome />} />  {/* 默认显示 */}
  <Route path="profile" element={<Profile />} />
  <Route path="settings" element={<Settings />} />
</Route>

// 访问 /dashboard 时，显示 DashboardHome
// 访问 /dashboard/profile 时，显示 Profile
```

## 路由守卫（权限控制）

### 什么是路由守卫？

路由守卫用于控制用户是否有权限访问某个路由，常用于实现登录验证、权限控制等。

### 实现登录验证

```typescript
import { Navigate, useLocation } from 'react-router-dom'

// 1. 创建 ProtectedRoute 组件
function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const isAuthenticated = checkAuth()  // 检查是否登录
  const location = useLocation()

  if (!isAuthenticated) {
    // 未登录，重定向到登录页，并保存当前路径
    return <Navigate to="/login" state={{ from: location }} replace />
  }

  return <>{children}</>
}

// 2. 使用 ProtectedRoute 包裹需要保护的路由
function App() {
  return (
    <Routes>
      <Route path="/" element={<Home />} />
      <Route path="/login" element={<Login />} />

      {/* 需要登录才能访问 */}
      <Route
        path="/dashboard"
        element={
          <ProtectedRoute>
            <Dashboard />
          </ProtectedRoute>
        }
      />

      <Route
        path="/profile"
        element={
          <ProtectedRoute>
            <Profile />
          </ProtectedRoute>
        }
      />
    </Routes>
  )
}

// 3. 登录页面
function Login() {
  const navigate = useNavigate()
  const location = useLocation()

  const from = location.state?.from?.pathname || '/'

  const handleLogin = async () => {
    // 登录逻辑
    await login()
    // 登录成功，跳转到之前的页面
    navigate(from, { replace: true })
  }

  return (
    <div>
      <h1>登录</h1>
      <button onClick={handleLogin}>登录</button>
    </div>
  )
}

// 4. 检查登录状态
function checkAuth(): boolean {
  const token = localStorage.getItem('token')
  return !!token
}
```

### 实现权限控制

```typescript
// 1. 定义用户角色
type Role = 'admin' | 'user' | 'guest'

interface User {
  id: string
  name: string
  role: Role
}

// 2. 创建权限路由组件
function RoleBasedRoute({
  children,
  allowedRoles
}: {
  children: React.ReactNode
  allowedRoles: Role[]
}) {
  const user = getCurrentUser()  // 获取当前用户

  if (!user) {
    return <Navigate to="/login" replace />
  }

  if (!allowedRoles.includes(user.role)) {
    return <Navigate to="/403" replace />  // 无权限
  }

  return <>{children}</>
}

// 3. 使用
function App() {
  return (
    <Routes>
      <Route path="/" element={<Home />} />
      <Route path="/login" element={<Login />} />
      <Route path="/403" element={<Forbidden />} />

      {/* 只有管理员可以访问 */}
      <Route
        path="/admin"
        element={
          <RoleBasedRoute allowedRoles={['admin']}>
            <AdminPanel />
          </RoleBasedRoute>
        }
      />

      {/* 管理员和普通用户可以访问 */}
      <Route
        path="/dashboard"
        element={
          <RoleBasedRoute allowedRoles={['admin', 'user']}>
            <Dashboard />
          </RoleBasedRoute>
        }
      />
    </Routes>
  )
}

function Forbidden() {
  return (
    <div>
      <h1>403 - 无权限访问</h1>
      <Link to="/">回到首页</Link>
    </div>
  )
}
```

### 使用 Context 管理认证状态

```typescript
import { createContext, useContext, useState, ReactNode } from 'react'

// 1. 创建认证 Context
interface AuthContextType {
  user: User | null
  login: (user: User) => void
  logout: () => void
  isAuthenticated: boolean
}

const AuthContext = createContext<AuthContextType | undefined>(undefined)

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null)

  const login = (user: User) => {
    setUser(user)
    localStorage.setItem('user', JSON.stringify(user))
  }

  const logout = () => {
    setUser(null)
    localStorage.removeItem('user')
  }

  const isAuthenticated = !!user

  return (
    <AuthContext.Provider value={{ user, login, logout, isAuthenticated }}>
      {children}
    </AuthContext.Provider>
  )
}

export function useAuth() {
  const context = useContext(AuthContext)
  if (!context) {
    throw new Error('useAuth 必须在 AuthProvider 内使用')
  }
  return context
}

// 2. 使用 Context 的路由守卫
function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const { isAuthenticated } = useAuth()
  const location = useLocation()

  if (!isAuthenticated) {
    return <Navigate to="/login" state={{ from: location }} replace />
  }

  return <>{children}</>
}

// 3. 在应用中使用
function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route
            path="/dashboard"
            element={
              <ProtectedRoute>
                <Dashboard />
              </ProtectedRoute>
            }
          />
        </Routes>
      </BrowserRouter>
    </AuthProvider>
  )
}
```

## 路由懒加载（代码分割）

### 为什么需要懒加载？

**问题**：所有页面的代码都打包在一起，首次加载时间长

**解决方案**：按路由拆分代码，访问时才加载对应的代码

### 使用 React.lazy 和 Suspense

```typescript
import { lazy, Suspense } from 'react'
import { BrowserRouter, Routes, Route } from 'react-router-dom'

// 1. 使用 lazy 懒加载组件
const Home = lazy(() => import('./pages/Home'))
const About = lazy(() => import('./pages/About'))
const Dashboard = lazy(() => import('./pages/Dashboard'))

// 2. 用 Suspense 包裹，提供加载状态
function App() {
  return (
    <BrowserRouter>
      <Suspense fallback={<LoadingSpinner />}>
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/about" element={<About />} />
          <Route path="/dashboard" element={<Dashboard />} />
        </Routes>
      </Suspense>
    </BrowserRouter>
  )
}

// 3. 加载指示器
function LoadingSpinner() {
  return (
    <div className="loading">
      <div className="spinner"></div>
      <p>加载中...</p>
    </div>
  )
}
```

### 错误边界（Error Boundary）

```typescript
import { Component, ReactNode } from 'react'

// 1. 创建错误边界组件
class ErrorBoundary extends Component<
  { children: ReactNode },
  { hasError: boolean; error: Error | null }
> {
  constructor(props: { children: ReactNode }) {
    super(props)
    this.state = { hasError: false, error: null }
  }

  static getDerivedStateFromError(error: Error) {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: any) {
    console.error('路由错误:', error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="error-page">
          <h1>出错了</h1>
          <p>{this.state.error?.message}</p>
          <button onClick={() => window.location.href = '/'}>
            回到首页
          </button>
        </div>
      )
    }

    return this.props.children
  }
}

// 2. 使用错误边界
function App() {
  return (
    <BrowserRouter>
      <ErrorBoundary>
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/about" element={<About />} />
        </Routes>
      </ErrorBoundary>
    </BrowserRouter>
  )
}
```

import { lazy } from 'react'

const Home = lazy(() => import('./pages/Home'))
const About = lazy(() => import('./pages/About'))
const Dashboard = lazy(() => import('./pages/Dashboard'))

export const routes = [
  {
    path: '/',

    path: '/about',
    element: <About />,
    title: '关于'
  },

function App() {
  return (
    <BrowserRouter>
      <Suspense fallback={<LoadingSpinner />}>
        <Routes>
          {routes.map(route => (
            <Route
              key={route.path}
              path={route.path}
              element={
                route.protected ? (
                  <ProtectedRoute>{route.element}</ProtectedRoute>
                ) : (
                  route.element
                )

        </Routes>
      </Suspense>
    </BrowserRouter>
  )

## 总结

React Router 是 React 应用的路由管理库，支持嵌套路由、路由守卫、懒加载等高级功能。

**核心要点**：
- 嵌套路由用 `Outlet` 渲染子路由
- 路由守卫用于登录验证和权限控制
- 懒加载用 `React.lazy` 和 `Suspense`
- 集中管理路由配置，使用路径常量
