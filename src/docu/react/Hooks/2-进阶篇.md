## 一、useReducer：复杂状态管理

### 1.1 为什么需要 useReducer？

**useState 的局限**：
```typescript
// 多个相关状态，更新逻辑分散
function TodoApp() {
  const [todos, setTodos] = useState([]);
  const [filter, setFilter] = useState('all');
  const [editingId, setEditingId] = useState(null);

  const addTodo = (text) => {
    setTodos([...todos, { id: Date.now(), text, done: false }]);
  };

  const toggleTodo = (id) => {
    setTodos(todos.map(t => t.id === id ? { ...t, done: !t.done } : t));
  };

  const deleteTodo = (id) => {
    setTodos(todos.filter(t => t.id !== id));
  };

  // 更新逻辑越来越多...
}
```

**useReducer 的优势**：
- 多个相关状态集中管理
- 更新逻辑统一在 reducer 里
- 更容易测试（reducer 是纯函数）
- 方便传递 dispatch 给子组件

### 1.2 基本用法

```typescript
import { useReducer } from 'react';

// 1. 定义状态类型和 action 类型
type State = {
  count: number;
  step: number;
};

type Action =
  | { type: 'increment' }
  | { type: 'decrement' }
  | { type: 'reset' }
  | { type: 'setStep'; step: number };

// 2. 定义 reducer 函数
function counterReducer(state: State, action: Action): State {
  switch (action.type) {
    case 'increment':
      return { ...state, count: state.count + state.step };
    case 'decrement':
      return { ...state, count: state.count - state.step };
    case 'reset':
      return { ...state, count: 0 };
    case 'setStep':
      return { ...state, step: action.step };
    default:
      return state;
  }
}

// 3. 使用 useReducer
function Counter() {
  const [state, dispatch] = useReducer(counterReducer, {
    count: 0,
    step: 1
  });

  return (
    <div>
      <p>计数: {state.count}</p>
      <p>步长: {state.step}</p>

      <button onClick={() => dispatch({ type: 'increment' })}>+</button>
      <button onClick={() => dispatch({ type: 'decrement' })}>-</button>
      <button onClick={() => dispatch({ type: 'reset' })}>重置</button>

      <input
        type="number"
        value={state.step}
        onChange={e => dispatch({ type: 'setStep', step: +e.target.value })}
      />
    </div>
  );
}
```

**语法**：
```typescript
const [state, dispatch] = useReducer(reducer, initialState);
```

- `reducer`：`(state, action) => newState`
- `initialState`：初始状态
- 返回：`[当前状态, dispatch函数]`

### 1.3 实战案例：Todo 应用

```typescript
type Todo = {
  id: number;
  text: string;
  done: boolean;
};

type State = {
  todos: Todo[];
  filter: 'all' | 'active' | 'done';
};

type Action =
  | { type: 'add'; text: string }
  | { type: 'toggle'; id: number }
  | { type: 'delete'; id: number }
  | { type: 'edit'; id: number; text: string }
  | { type: 'setFilter'; filter: State['filter'] };

function todoReducer(state: State, action: Action): State {
  switch (action.type) {
    case 'add':
      return {
        ...state,
        todos: [...state.todos, { id: Date.now(), text: action.text, done: false }]
      };

    case 'toggle':
      return {
        ...state,
        todos: state.todos.map(t =>
          t.id === action.id ? { ...t, done: !t.done } : t
        )
      };

    case 'delete':
      return {
        ...state,
        todos: state.todos.filter(t => t.id !== action.id)
      };

    case 'edit':
      return {
        ...state,
        todos: state.todos.map(t =>
          t.id === action.id ? { ...t, text: action.text } : t
        )
      };

    case 'setFilter':
      return { ...state, filter: action.filter };

    default:
      return state;
  }
}

function TodoApp() {
  const [state, dispatch] = useReducer(todoReducer, {
    todos: [],
    filter: 'all'
  });

  const [input, setInput] = useState('');

  const handleAdd = () => {
    if (input.trim()) {
      dispatch({ type: 'add', text: input });
      setInput('');
    }
  };

  const filteredTodos = state.todos.filter(t => {
    if (state.filter === 'active') return !t.done;
    if (state.filter === 'done') return t.done;
    return true;
  });

  return (
    <div>
      <input
        value={input}
        onChange={e => setInput(e.target.value)}
        onKeyDown={e => e.key === 'Enter' && handleAdd()}
      />
      <button onClick={handleAdd}>添加</button>

      <div>
        {(['all', 'active', 'done'] as const).map(f => (
          <button
            key={f}
            onClick={() => dispatch({ type: 'setFilter', filter: f })}
            style={{ fontWeight: state.filter === f ? 'bold' : 'normal' }}
          >
            {f}
          </button>
        ))}
      </div>

      <ul>
        {filteredTodos.map(todo => (
          <li key={todo.id}>
            <input
              type="checkbox"
              checked={todo.done}
              onChange={() => dispatch({ type: 'toggle', id: todo.id })}
            />
            <span style={{ textDecoration: todo.done ? 'line-through' : 'none' }}>
              {todo.text}
            </span>
            <button onClick={() => dispatch({ type: 'delete', id: todo.id })}>删除</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
```

### 1.4 useReducer vs useState

| 维度 | useState | useReducer |
|------|----------|------------|
| **复杂度** | 简单状态（1-2个值） | 复杂状态（对象、多个相关状态） |
| **更新逻辑** | 分散在各处 | 集中在 reducer |
| **可测试性** | 难测试（依赖组件） | 易测试（纯函数） |
| **类型提示** | 一般 | 更好（通过 Action 类型） |
| **性能** | 差不多 | 略好（减少闭包捕获） |

**选择建议**：
- **简单状态**：用 `useState`（如开关、输入框）
- **多个相关状态**：用 `useReducer`（如表单、购物车）
- **状态更新复杂**：用 `useReducer`（如 state 依赖上一次 state）

## 二、useMemo：缓存计算结果

### 2.1 什么时候需要 useMemo？

**问题场景**：
```typescript
function ProductList({ products, filter }) {
  // ❌ 每次渲染都重新计算，即使 products 和 filter 没变
  const filteredProducts = products.filter(p => p.category === filter);

  return (
    <ul>
      {filteredProducts.map(p => <li key={p.id}>{p.name}</li>)}
    </ul>
  );
}
```

**解决**：
```typescript
function ProductList({ products, filter }) {
  // ✅ 只在 products 或 filter 变化时重新计算
  const filteredProducts = useMemo(() => {
    console.log('重新过滤');
    return products.filter(p => p.category === filter);
  }, [products, filter]);

  return (
    <ul>
      {filteredProducts.map(p => <li key={p.id}>{p.name}</li>)}
    </ul>
  );
}
```

### 2.2 基本用法

```typescript
const memoizedValue = useMemo(() => {
  // 复杂计算
  return computeExpensiveValue(a, b);
}, [a, b]); // 依赖数组
```

- **第一个参数**：计算函数（返回需要缓存的值）
- **第二个参数**：依赖数组（依赖变化时重新计算）
- **返回**：缓存的值

### 2.3 典型使用场景

#### 场景一：复杂计算

```typescript
function PrimeNumbers({ max }: { max: number }) {
  const primes = useMemo(() => {
    console.log('计算质数');
    const result: number[] = [];

    for (let i = 2; i <= max; i++) {
      let isPrime = true;
      for (let j = 2; j <= Math.sqrt(i); j++) {
        if (i % j === 0) {
          isPrime = false;
          break;
        }
      }
      if (isPrime) result.push(i);
    }

    return result;
  }, [max]); // 只在 max 变化时重新计算

  return <div>质数: {primes.join(', ')}</div>;
}
```

#### 场景二：避免子组件无效渲染

```typescript
// 子组件用了 React.memo
const ExpensiveChild = React.memo(({ data }: { data: any }) => {
  console.log('ExpensiveChild 渲染');
  return <div>{JSON.stringify(data)}</div>;
});

function Parent() {
  const [count, setCount] = useState(0);

  // ❌ 每次父组件渲染，data 都是新对象，子组件也会重渲染
  const data = { value: 100 };

  // ✅ data 引用不变，子组件不会重渲染
  const memoizedData = useMemo(() => ({ value: 100 }), []);

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>父组件计数: {count}</button>
      <ExpensiveChild data={memoizedData} />
    </div>
  );
}
```

### 2.4 常见误区

#### 误区一：过度使用

```typescript
// ❌ 不要：简单计算用 useMemo 反而更慢
const doubled = useMemo(() => count * 2, [count]); // Hook 本身也有开销

// ✅ 正确：直接计算
const doubled = count * 2;
```

**何时才用 useMemo？**
- 计算量大（如排序、过滤大数组）
- 创建复杂对象/数组，且传给用了 `memo` 的子组件
- 作为其他 Hook 的依赖（避免无限循环）

#### 误区二：忘记依赖

```typescript
function Bad({ userId, limit }) {
  // ❌ 依赖数组不完整
  const query = useMemo(() => {
    return { userId, limit }; // 用了 userId 和 limit
  }, [userId]); // 但只写了 userId

  // limit 变化时 query 不会更新！
}
```

**解决**：让 ESLint 插件自动检查依赖。

## 三、useCallback：缓存函数引用

### 3.1 为什么需要 useCallback？

**问题**：每次渲染都创建新函数，导致子组件无效渲染。

```typescript
const Child = React.memo(({ onClick }: { onClick: () => void }) => {
  console.log('Child 渲染');
  return <button onClick={onClick}>点击</button>;
});

function Parent() {
  const [count, setCount] = useState(0);

  // ❌ 每次渲染都是新函数，Child 每次都重渲染
  const handleClick = () => {
    console.log('点击了');
  };

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>父组件计数: {count}</button>
      <Child onClick={handleClick} />
    </div>
  );
}
```

**解决**：
```typescript
function Parent() {
  const [count, setCount] = useState(0);

  // ✅ 缓存函数引用
  const handleClick = useCallback(() => {
    console.log('点击了');
  }, []); // 依赖为空，函数永远不变

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>父组件计数: {count}</button>
      <Child onClick={handleClick} />
    </div>
  );
}
```

### 3.2 基本用法

```typescript
const memoizedCallback = useCallback(
  () => {
    doSomething(a, b);
  },
  [a, b] // 依赖数组
);
```

- **等价于**：`useMemo(() => () => doSomething(a, b), [a, b])`
- **返回**：缓存的函数引用

### 3.3 典型使用场景

#### 场景一：传给用了 memo 的子组件

```typescript
const TodoItem = React.memo(({ todo, onToggle, onDelete }) => {
  console.log('TodoItem 渲染:', todo.id);
  return (
    <li>
      <input type="checkbox" checked={todo.done} onChange={onToggle} />
      {todo.text}
      <button onClick={onDelete}>删除</button>
    </li>
  );
});

function TodoList() {
  const [todos, setTodos] = useState([
    { id: 1, text: '学习 React', done: false },
    { id: 2, text: '写代码', done: true }
  ]);

  // ✅ 缓存函数，避免所有 TodoItem 重渲染
  const handleToggle = useCallback((id: number) => {
    setTodos(todos => todos.map(t =>
      t.id === id ? { ...t, done: !t.done } : t
    ));
  }, []); // 依赖为空（使用函数式更新）

  const handleDelete = useCallback((id: number) => {
    setTodos(todos => todos.filter(t => t.id !== id));
  }, []);

  return (
    <ul>
      {todos.map(todo => (
        <TodoItem
          key={todo.id}
          todo={todo}
          onToggle={() => handleToggle(todo.id)}
          onDelete={() => handleDelete(todo.id)}
        />
      ))}
    </ul>
  );
}
```

#### 场景二：作为 useEffect 的依赖

```typescript
function DataFetcher({ userId }) {
  const [data, setData] = useState(null);

  // 缓存函数，避免 useEffect 无限循环
  const fetchData = useCallback(async () => {
    const res = await fetch(`/api/user/${userId}`);
    const data = await res.json();
    setData(data);
  }, [userId]); // userId 变化时函数才变

  useEffect(() => {
    fetchData();
  }, [fetchData]); // 依赖 fetchData

  return <div>{data?.name}</div>;
}
```

### 3.4 useCallback vs useMemo

| Hook | 缓存的是什么 | 等价写法 |
|------|--------------|----------|
| `useCallback` | 函数本身 | `useMemo(() => fn, deps)` |
| `useMemo` | 函数返回值 | - |

```typescript
// 这两个等价
const memoizedCallback = useCallback(() => doSomething(), [a, b]);
const memoizedCallback = useMemo(() => () => doSomething(), [a, b]);

// useMemo 缓存结果
const memoizedValue = useMemo(() => doSomething(), [a, b]);
```

## 四、React.memo：组件级缓存

### 4.1 什么时候用 React.memo？

**问题**：父组件重渲染，子组件也会重渲染（即使 props 没变）。

```typescript
function Child({ name }: { name: string }) {
  console.log('Child 渲染');
  return <div>{name}</div>;
}

function Parent() {
  const [count, setCount] = useState(0);

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>计数: {count}</button>
      <Child name="Tom" />
      {/* count 变化时，Child 也会重渲染（虽然 name 没变） */}
    </div>
  );
}
```

**解决**：
```typescript
// 用 React.memo 包裹
const Child = React.memo(({ name }: { name: string }) => {
  console.log('Child 渲染');
  return <div>{name}</div>;
});

// 现在 count 变化时，Child 不会重渲染（因为 name 没变）
```

### 4.2 基本用法

```typescript
const MemoizedComponent = React.memo(Component);

// 自定义比较函数（很少用）
const MemoizedComponent = React.memo(Component, (prevProps, nextProps) => {
  // 返回 true 表示不需要重渲染
  return prevProps.id === nextProps.id;
});
```

**默认行为**：**浅比较** props（用 `Object.is`）。

### 4.3 配合 useMemo / useCallback

**错误示范**：
```typescript
const Child = React.memo(({ data, onClick }) => {
  console.log('Child 渲染');
  return <button onClick={onClick}>{data.value}</button>;
});

function Parent() {
  const [count, setCount] = useState(0);

  // ❌ 每次渲染都是新对象/新函数
  const data = { value: 100 };
  const handleClick = () => console.log('点击');

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>计数: {count}</button>
      <Child data={data} onClick={handleClick} />
      {/* Child 还是每次都重渲染（因为 data 和 onClick 引用变了） */}
    </div>
  );
}
```

**正确做法**：
```typescript
function Parent() {
  const [count, setCount] = useState(0);

  // ✅ 缓存对象和函数
  const data = useMemo(() => ({ value: 100 }), []);
  const handleClick = useCallback(() => console.log('点击'), []);

  return (
    <div>
      <button onClick={() => setCount(count + 1)}>计数: {count}</button>
      <Child data={data} onClick={handleClick} />
      {/* 现在 Child 不会重渲染了 */}
    </div>
  );
}
```

**总结**：
- `React.memo` + `useMemo` / `useCallback` = 完整优化
- 只用 `React.memo` 但传了对象/函数 = 白优化

## 五、自定义 Hooks

### 5.1 什么是自定义 Hooks？

**定义**：以 `use` 开头的函数，内部可以调用其他 Hooks，用于**复用状态逻辑**。

**和普通函数的区别**：
- 自定义 Hooks **可以调用** Hooks（如 `useState`、`useEffect`）
- 普通函数**不能**调用 Hooks

### 5.2 基本示例

```typescript
// 自定义 Hook：管理输入框
function useInput(initialValue: string) {
  const [value, setValue] = useState(initialValue);

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setValue(e.target.value);
  };

  const reset = () => setValue(initialValue);

  return {
    value,
    onChange: handleChange,
    reset
  };
}

// 使用
function LoginForm() {
  const username = useInput('');
  const password = useInput('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    console.log(username.value, password.value);
    username.reset();
    password.reset();
  };

  return (
    <form onSubmit={handleSubmit}>
      <input {...username} placeholder="用户名" />
      <input {...password} type="password" placeholder="密码" />
      <button type="submit">登录</button>
    </form>
  );
}
```

### 5.3 实战案例

#### 案例一：useFetch（数据请求）

```typescript
interface UseFetchResult<T> {
  data: T | null;
  loading: boolean;
  error: string | null;
  refetch: () => void;
}

function useFetch<T>(url: string): UseFetchResult<T> {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchData = useCallback(async () => {
    setLoading(true);
    setError(null);

    try {
      const res = await fetch(url);
      if (!res.ok) throw new Error(`HTTP ${res.status}`);

      const json = await res.json();
      setData(json);
    } catch (err) {
      setError(err instanceof Error ? err.message : '未知错误');
    } finally {
      setLoading(false);
    }
  }, [url]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  return { data, loading, error, refetch: fetchData };
}

// 使用
function UserList() {
  const { data, loading, error, refetch } = useFetch<User[]>('/api/users');

  if (loading) return <div>加载中...</div>;
  if (error) return <div>错误: {error}</div>;

  return (
    <div>
      <button onClick={refetch}>刷新</button>
      <ul>
        {data?.map(user => <li key={user.id}>{user.name}</li>)}
      </ul>
    </div>
  );
}
```

#### 案例二：useLocalStorage（持久化状态）

```typescript
function useLocalStorage<T>(key: string, initialValue: T) {
  // 惰性初始化：从 localStorage 读取
  const [value, setValue] = useState<T>(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(error);
      return initialValue;
    }
  });

  // 更新 localStorage
  const setStoredValue = (newValue: T | ((val: T) => T)) => {
    try {
      const valueToStore = newValue instanceof Function ? newValue(value) : newValue;
      setValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catch (error) {
      console.error(error);
    }
  };

  return [value, setStoredValue] as const;
}

// 使用
function ThemeToggle() {
  const [theme, setTheme] = useLocalStorage('theme', 'light');

  return (
    <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
      当前主题: {theme}
    </button>
  );
}
```

#### 案例三：useDebounce（防抖）

```typescript
function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

// 使用
function SearchBox() {
  const [query, setQuery] = useState('');
  const debouncedQuery = useDebounce(query, 500); // 延迟 500ms

  useEffect(() => {
    if (debouncedQuery) {
      fetch(`/api/search?q=${debouncedQuery}`)
        .then(res => res.json())
        .then(console.log);
    }
  }, [debouncedQuery]); // 只在防抖后的值变化时请求

  return (
    <input
      value={query}
      onChange={e => setQuery(e.target.value)}
      placeholder="搜索..."
    />
  );
}
```

#### 案例四：useToggle（开关状态）

```typescript
function useToggle(initialValue: boolean = false) {
  const [value, setValue] = useState(initialValue);

  const toggle = useCallback(() => setValue(v => !v), []);
  const setTrue = useCallback(() => setValue(true), []);
  const setFalse = useCallback(() => setValue(false), []);

  return { value, toggle, setTrue, setFalse };
}

// 使用
function Modal() {
  const modal = useToggle();

  return (
    <div>
      <button onClick={modal.setTrue}>打开弹窗</button>

      {modal.value && (
        <div className="modal">
          <p>这是一个弹窗</p>
          <button onClick={modal.setFalse}>关闭</button>
        </div>
      )}
    </div>
  );
}
```

### 5.4 自定义 Hooks 的最佳实践

#### 规则一：必须以 use 开头

```typescript
// ✅ 正确
function useAuth() { /* ... */ }

// ❌ 错误（React 不会检查 Hooks 规则）
function getAuth() {
  const [user, setUser] = useState(null); // 会报错
}
```

#### 规则二：返回值的设计

```typescript
// 方式一：数组（类似 useState）
function useToggle() {
  const [value, setValue] = useState(false);
  const toggle = () => setValue(v => !v);
  return [value, toggle] as const;
}
const [isOpen, toggleOpen] = useToggle(); // 可以自由命名

// 方式二：对象（字段多时推荐）
function useAuth() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(false);

  const login = async (username, password) => { /* ... */ };
  const logout = () => setUser(null);

  return { user, loading, login, logout };
}
const { user, login, logout } = useAuth(); // 字段名固定
```

**选择建议**：
- **1-2 个返回值**：用数组（方便重命名）
- **3+ 个返回值**：用对象（避免记忆顺序）

#### 规则三：避免过度封装

```typescript
// ❌ 不要：简单逻辑不需要自定义 Hook
function useCount(initialValue: number) {
  const [count, setCount] = useState(initialValue);
  return [count, setCount];
}
// 直接用 useState 就行

// ✅ 正确：有明确的复用价值
function useCounter(initialValue: number = 0, step: number = 1) {
  const [count, setCount] = useState(initialValue);

  const increment = () => setCount(c => c + step);
  const decrement = () => setCount(c => c - step);
  const reset = () => setCount(initialValue);

  return { count, increment, decrement, reset };
}
```

## 六、React 18 并发特性

### 6.1 useTransition：低优先级更新

**问题**：输入框打字时，同步更新大列表会卡顿。

```typescript
function SearchPage() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);

  const handleChange = (e) => {
    const newQuery = e.target.value;
    setQuery(newQuery); // 输入框更新

    // 过滤 10000 条数据（很慢！）
    setResults(hugeList.filter(item => item.includes(newQuery)));
    // 导致输入卡顿
  };

  return (
    <div>
      <input value={query} onChange={handleChange} />
      <ResultList results={results} /> {/* 10000 个元素 */}
    </div>
  );
}
```

**解决**：用 `useTransition` 标记低优先级更新。

```typescript
import { useState, useTransition } from 'react';

function SearchPage() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);
  const [isPending, startTransition] = useTransition();

  const handleChange = (e) => {
    const newQuery = e.target.value;
    setQuery(newQuery); // 高优先级：立即更新输入框

    // 低优先级：不阻塞输入
    startTransition(() => {
      setResults(hugeList.filter(item => item.includes(newQuery)));
    });
  };

  return (
    <div>
      <input value={query} onChange={handleChange} />
      {isPending && <div>搜索中...</div>}
      <ResultList results={results} />
    </div>
  );
}
```

**效果**：
- 输入框立即响应（高优先级）
- 列表更新可以被打断（低优先级）
- `isPending` 显示加载状态

### 6.2 useDeferredValue：延迟更新

**和 useTransition 的区别**：
- `useTransition`：主动标记更新为低优先级
- `useDeferredValue`：被动获取「延迟版本」的值

```typescript
function SearchPage() {
  const [query, setQuery] = useState('');

  // deferredQuery 会"延迟"跟上 query
  const deferredQuery = useDeferredValue(query);

  // 用延迟的值去过滤（不阻塞输入）
  const results = useMemo(() => {
    return hugeList.filter(item => item.includes(deferredQuery));
  }, [deferredQuery]);

  return (
    <div>
      <input
        value={query}
        onChange={e => setQuery(e.target.value)}
      />
      <ResultList results={results} />
    </div>
  );
}
```

**何时用哪个？**
- **能控制 setState**：用 `useTransition`
- **只能用传入的值**（如 props）：用 `useDeferredValue`

### 6.3 useId：生成唯一 ID

**问题**：SSR 时客户端和服务端生成的随机 ID 不一致，导致 hydration 错误。

```typescript
// ❌ 不好：每次渲染 ID 可能不同
function Form() {
  const id = Math.random().toString();
  return (
    <div>
      <label htmlFor={id}>用户名</label>
      <input id={id} />
    </div>
  );
}
```

**解决**：
```typescript
import { useId } from 'react';

function Form() {
  const id = useId(); // 生成唯一且稳定的 ID

  return (
    <div>
      <label htmlFor={id}>用户名</label>
      <input id={id} />
    </div>
  );
}
```

**特点**：
- 客户端和服务端生成的 ID **相同**
- 同一个组件的多个实例 ID **不同**
- 不适合做列表的 `key`（用数据的唯一标识）

## 七、Hooks 的进阶技巧

### 7.1 组合多个自定义 Hooks

```typescript
function useUser(userId: string) {
  const { data, loading, error } = useFetch<User>(`/api/users/${userId}`);
  const [favorites, setFavorites] = useLocalStorage<number[]>('favorites', []);

  const isFavorite = favorites.includes(data?.id ?? -1);

  const toggleFavorite = () => {
    if (!data) return;

    setFavorites(prev =>
      prev.includes(data.id)
        ? prev.filter(id => id !== data.id)
        : [...prev, data.id]
    );
  };

  return { user: data, loading, error, isFavorite, toggleFavorite };
}

// 使用
function UserProfile({ userId }) {

  return (
    <div>
      <h1>{user?.name}</h1>
      <button onClick={toggleFavorite}>
        {isFavorite ? '取消收藏' : '收藏'}
      </button>
    </div>
  );
}
```

### 7.2 使用 useReducer + Context（轻量状态管理）

```typescript
// 1. 定义 Context
type AuthState = {
  user: User | null;
  loading: boolean;
};

type AuthAction =
  | { type: 'login'; user: User }
  | { type: 'logout' };

const AuthContext = createContext<{
  state: AuthState;
  dispatch: React.Dispatch<AuthAction>;
} | null>(null);

// 2. Reducer
function authReducer(state: AuthState, action: AuthAction): AuthState {
  switch (action.type) {
    case 'login':
      return { user: action.user, loading: false };
    case 'logout':
      return { user: null, loading: false };
    default:
      return state;
  }
}

// 3. Provider 组件
function AuthProvider({ children }: { children: React.ReactNode }) {
  const [state, dispatch] = useReducer(authReducer, {
    user: null,
    loading: false
  });

  return (
    <AuthContext.Provider value={{ state, dispatch }}>
      {children}
    </AuthContext.Provider>
  );
}

// 4. 自定义 Hook
function useAuth() {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth 必须在 AuthProvider 内使用');
  return context;
}

// 5. 使用
function App() {
  return (
    <AuthProvider>
      <Navbar />
      <Content />
    </AuthProvider>
  );
}

function Navbar() {
  const { state, dispatch } = useAuth();

  return (
    <div>
      {state.user ? (
        <>
          <span>欢迎, {state.user.name}</span>
          <button onClick={() => dispatch({ type: 'logout' })}>登出</button>
        </>
      ) : (
        <button onClick={() => dispatch({ type: 'login', user: mockUser })}>
          登录
        </button>
      )}
    </div>
  );
}
```

## 八、面试检验题

### 题目 1：useCallback 和 useMemo 的区别？

**答案要点**：
1. **缓存的内容**：
   - `useCallback`：缓存函数本身
   - `useMemo`：缓存函数的返回值
2. **等价写法**：`useCallback(fn, deps)` === `useMemo(() => fn, deps)`
3. **使用场景**：
   - `useCallback`：传给用了 `memo` 的子组件、作为 `useEffect` 依赖
   - `useMemo`：复杂计算、缓存对象/数组避免子组件重渲染

### 题目 2：什么时候用 useReducer 代替 useState？

**答案要点**：
1. **多个相关状态**：如表单的多个字段
2. **复杂更新逻辑**：下一个状态依赖上一个状态
3. **更新逻辑需要复用**：reducer 是纯函数，易测试
4. **传递更新方法给子组件**：传 `dispatch` 比传多个 `setState` 更简洁

### 题目 3：自定义 Hooks 的命名规则是什么？为什么？

**答案要点**：
1. **规则**：必须以 `use` 开头
2. **原因**：React 通过命名识别是否是 Hook，从而检查 Hooks 规则
3. **后果**：不以 `use` 开头，ESLint 不会检查，可能违反 Hooks 规则

### 题目 4：React.memo 什么时候会失效？

**答案要点**：
1. **传入的 props 是引用类型**：每次渲染都是新对象/数组/函数
2. **解决**：用 `useMemo` 缓存对象/数组，用 `useCallback` 缓存函数
3. **误区**：只用 `React.memo` 但不处理 props，等于没优化

### 题目 5：useTransition 和 useDeferredValue 的区别？

**答案要点**：
1. **useTransition**：主动标记更新为低优先级，返回 `isPending` 状态
2. **useDeferredValue**：被动获取延迟的值，适合无法控制 setState 的场景
3. **选择**：能控制 setState 用 `useTransition`，只能用传入的值用 `useDeferredValue`

## 总结

### 核心 Hooks 对比

| Hook | 用途 | 何时使用 |
|------|------|----------|
| `useReducer` | 复杂状态管理 | 多个相关状态、复杂更新逻辑 |
| `useMemo` | 缓存计算结果 | 复杂计算、避免子组件无效渲染 |
| `useCallback` | 缓存函数引用 | 传给用了 memo 的子组件 |
| `useTransition` | 低优先级更新 | 大量计算不阻塞 UI |
| `useDeferredValue` | 延迟值 | 无法控制 setState 的场景 |

### 性能优化三板斧

1. **React.memo**：阻止子组件无效渲染
2. **useMemo / useCallback**：稳定 props 的引用
3. **useTransition**：大量计算不阻塞 UI

**记住**：不要过早优化，先写代码，有性能问题再优化！
